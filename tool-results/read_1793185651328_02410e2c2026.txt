    37→    37→export default function DynamicIsland() {
    38→    38→  const mounted = useMounted();
    39→    39→
    40→    40→  const expanded = useIsland((s) => s.expanded);
    41→    41→  const alert = useIsland((s) => s.alert);
    42→    42→
    43→    43→  // Auto-dismiss island alerts after ~4.5s.
    44→    44→  useEffect(() => {
    45→    45→    if (!alert) return;
    46→    46→    const id = window.setTimeout(() => useIsland.getState().clearAlert(), 4500);
    47→    47→    return () => window.clearTimeout(id);
    48→    48→  }, [alert]);
    49→    49→
    50→    50→  // Cross-store notification subscriptions (task completions, timer start/stop).
    51→    51→  useEffect(() => {
    52→    52→    const countDone = (tasks: readonly { status: string }[]) =>
    53→    53→      tasks.reduce((n, t) => (t.status === 'done' ? n + 1 : n), 0);
    54→    54→
    55→    55→    const unsubNexus = useNexus.subscribe((s, prev) => {
    56→    56→      if (countDone(s.tasks) > countDone(prev.tasks)) {
    57→    57→        useIsland.getState().notify({ title: 'Task completed', detail: '+1 task cleared', tone: 'success' });
    58→    58→      }
    59→    59→    });
    60→    60→
    61→    61→    const unsubUI = useUI.subscribe((s, prev) => {
    62→    62→      const t = s.timer;
    63→    63→      const p = prev.timer;
    64→    64→      // Timer started from a fully idle state.
    65→    65→      if (t.running && !p.running && p.subjectId === null && p.accumulatedMs === 0) {
    66→    66→        const name = useNexus.getState().subjects.find((x) => x.id === t.subjectId)?.name ?? 'Free session';
    67→    67→        useIsland.getState().notify({ title: 'Focus started', detail: name, tone: 'info' });
    68→    68→      }
    69→    69→      // Timer stopped (full reset — pause keeps subjectId / accumulatedMs).
    70→    70→      if (p.running && !t.running && t.subjectId === null && t.startedAt === null && t.accumulatedMs === 0) {
    71→    71→        const minutes = Math.max(1, Math.round(timerElapsedMs(p) / 60000));
    72→    72→        useIsland.getState().notify({
    73→    73→          title: 'Session logged',
    74→    74→          detail: `${minutesToHM(minutes)} focused`,
    75→    75→          tone: 'success',
    76→    76→        });
    77→    77→      }
    78→    78→    });
    79→    79→
    80→    80→    return () => {
    81→    81→      unsubNexus();
    82→    82→      unsubUI();
    83→    83→    };
    84→    84→  }, []);
    85→    85→
    86→    86→  // Escape closes the island while expanded.
    87→    87→  useEffect(() => {
    88→    88→    if (!expanded) return;
    89→    89→    const onKey = (e: KeyboardEvent) => {
    90→    90→      if (e.key === 'Escape') useIsland.getState().collapse();
    91→    91→    };
    92→    92→    window.addEventListener('keydown', onKey);
    93→    93→    return () => window.removeEventListener('keydown', onKey);
    94→    94→  }, [expanded]);
    95→    95→
    96→    96→  // Music playing flag: shared by compact pill (eq) + music module (waveform).
    97→    97→  const [musicPlaying, setMusicPlaying] = useState(false);
    98→    98→  useEffect(() => {
    99→    99→    const id = window.setInterval(() => setMusicPlaying(lofiEngine.isPlaying()), 1000); // safety poll
   100→   100→    return () => window.clearInterval(id);
   101→   101→  }, []);
   102→   102→
   103→   103→  if (!mounted) return null; // hydration guard (persisted island/pomo state)
   104→   104→
   105→   105→  return (
   106→   106→    <>
   107→   107→      {expanded && (
   108→   108→        <div
   109→   109→          className="pointer-events-auto fixed inset-0 z-40"
   110→   110→          onClick={() => useIsland.getState().collapse()}
   111→   111→          aria-hidden="true"
   112→   112→        />
   113→   113→      )}
   114→   114→      <div className="pointer-events-none fixed inset-x-0 top-2.5 z-[60] flex flex-col items-center">
   115→   115→        {expanded ? (
   116→   116→          <ExpandedPanel musicPlaying={musicPlaying} onMusicChange={setMusicPlaying} />
   117→   117→        ) : (
   118→   118→          <CompactPill musicPlaying={musicPlaying} />
   119→   119→        )}
   120→   120→      </div>
   121→   121→    </>
   122→   122→  );
   123→   123→}

   126→   126→function CompactPill({ musicPlaying }: { musicPlaying: boolean }) {
   127→   127→  const alert = useIsland((s) => s.alert);
   128→   128→  const pomo = useIsland((s) => s.pomo);
   129→   129→  const timer = useUI((s) => s.timer);
   130→   130→  const subjects = useNexus((s) => s.subjects);
   131→   131→  const dailyLogs = useNexus((s) => s.dailyLogs);
   132→   132→  const habits = useNexus((s) => s.habits);
   133→   133→  const habitRecords = useNexus((s) => s.habitRecords);
   134→   134→
   135→   135→  const timerActive = timer.subjectId !== null || timer.accumulatedMs > 0 || timer.running;
   136→   136→  const pomoActive = pomo.phase !== 'idle';
   137→   137→  const now = useTimerTick(timerActive || pomoActive);
   138→   138→  const pomoLeft = pomoRemainingMs(pomo, now);
   139→   139→
   140→   140→  const todayMin = useMemo(() => dayTotalMinutes(dailyLogs[todayKey()]), [dailyLogs]);
   141→   141→  const habitScore = useMemo(() => {
   142→   142→    const active = habits.filter((h) => !h.archived);
   143→   143→    const key = todayKey();
   144→   144→    const done = active.reduce((n, h) => (habitRecords[h.id]?.[key] ? n + 1 : n), 0);
   145→   145→    return `${done}/${active.length}`;
   146→   146→  }, [habits, habitRecords]);
   147→   147→  const subjectLabel = useMemo(
   148→   148→    () => subjects.find((x) => x.id === timer.subjectId)?.name ?? '',
   149→   149→    [subjects, timer.subjectId],
   150→   150→  );
   151→   151→
   152→   152→  // Pulse the pill whenever a new alert arrives.
   153→   153→  const pulse = useAnimationControls();
   154→   154→  const alertId = alert?.id ?? 0;
   155→   155→  useEffect(() => {
   156→   156→    if (!alertId) return;
   157→   157→    void pulse.start({ scale: [1, 1.06, 1], transition: { duration: 0.55, times: [0, 0.35, 1], ease: 'easeOut' } });
   158→   158→  }, [alertId, pulse]);

   160→   160→  const contentKey = alert ? 'alert' : timerActive ? 'timer' : pomoActive ? 'pomo' : musicPlaying ? 'music' : 'idle';

   169→   169→      whileTap={{ scale: 0.97 }}

   172→   172→      aria-label="Open Dynamic Island"
   173→   173→    >
   174→   174→      <AnimatePresence mode="wait" initial={false}>
   175→   175→        <motion.span
   176→   176→          key={contentKey}
   177→   177→          initial={{ opacity: 0, y: 5 }}
   178→   178→          animate={{ opacity: 1, y: 0 }}
   179→   179→          exit={{ opacity: 0, y: -5 }}
   180→   180→          transition={{ duration: 0.14 }}
   181→   181→          className="flex min-w-0 items-center gap-2"
   182→   182→        >
   183→   183→          {alert ? (
   184→   184→            <>
   185→   185→              <span
   186→   186→                className={`island-live-dot h-1.5 w-1.5 shrink-0 rounded-full ${
   187→   187→                  alert.tone === 'success' ? 'bg-emerald-400' : alert.tone === 'warning' ? 'bg-amber-400' : 'bg-[var(--primary)]'
   188→   188→                }`}
   189→   189→              />
   190→   190→              <span className="max-w-[220px] truncate text-xs font-medium">{alert.title}</span>
   191→   191→            </>
   192→   192→          ) : timerActive ? (
   193→   193→            <>
   194→   194→              <span
   195→   195→                className={`h-1.5 w-1.5 shrink-0 rounded-full ${
   196→   196→                  timer.running ? 'island-live-dot animate-pulse bg-[var(--primary)]' : 'bg-white/40'
   197→   197→                }`}
   198→   198→              />
   199→   199→              <span className="font-mono text-[11px] tabular-nums">{secondsToClock(timerElapsedMs(timer) / 1000)}</span>
   200→   200→              {subjectLabel && <span className="max-w-[80px] truncate text-[10px] text-white/50">{subjectLabel}</span>}

   204→   204→              <span
   205→   205→                className={`island-live-dot animate-pulse h-1.5 w-1.5 shrink-0 rounded-full ${
   206→   206→                  pomo.phase === 'focus' ? 'bg-emerald-400' : 'bg-amber-400'
   207→   207→                }`}
   208→   208→              />
   209→   209→              <span className="font-mono text-[11px] tabular-nums">{fmtMs(pomoLeft)}</span>
   210→   210→              <span className="text-[10px] text-white/55">

   325→   325→function PomoActiveView({ pomoLeft }: { pomoLeft: number }) {

   327→   327→  const subjects = useNexus((s) => s.subjects);
   328→   328→

   331→   331→  const R = 44;
   332→   332→  const CIRC = 2 * Math.PI * R;
   333→   333→  const ringColor = pomo.phase === 'focus' ? 'var(--aur-3)' : '#fbbf24';
   334→   334→  const subjectName = subjects.find((x) => x.id === pomo.subjectId)?.name ?? 'First subject';
   335→   335→  const filled = pomo.cycles % 4;
   336→   336→  const paused = pomo.pausedRemainingMs !== null;
   337→   337→
   338→   338→  return (

   340→   340→      <div className="relative h-28 w-28 shrink-0">
   341→   341→        <svg viewBox="0 0 100 100" className="h-full w-full -rotate-90" aria-hidden="true">
   342→   342→          <circle cx="50" cy="50" r={R} fill="none" stroke="rgba(255,255,255,0.09)" strokeWidth="6" />

   344→   344→            cx="50" cy="50" r={R} fill="none" stroke={ringColor} strokeWidth="6" strokeLinecap="round"
   345→   345→            strokeDasharray={CIRC} strokeDashoffset={CIRC * (1 - fraction)}
   346→   346→            style={{ transition: 'stroke-dashoffset 0.6s linear' }}
   347→   347→          />
   348→   348→        </svg>

   623→   623→function MusicModule({ playing, onPlayingChange }: { playing: boolean; onPlayingChange: (v: boolean) => void }) {
   624→   624→  const [volume, setVolume] = useState(lofiEngine.getVolume());
   625→   625→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   626→   626→
   627→   627→  // Live oscilloscope — rAF only while playing.
   628→   628→  useEffect(() => {
   629→   629→    if (!playing) return;
   630→   630→    const canvas = canvasRef.current;
   631→   631→    const c2d = canvas?.getContext('2d');
   632→   632→    if (!canvas || !c2d) return;
   633→   633→    const dpr = window.devicePixelRatio || 1;
   634→   634→    const w = canvas.clientWidth;
   635→   635→    const h = canvas.clientHeight;
   636→   636→    canvas.width = Math.max(1, Math.floor(w * dpr));
   637→   637→    canvas.height = Math.max(1, Math.floor(h * dpr));
   638→   638→    c2d.scale(dpr, dpr);
   639→   639→    const primary = getComputedStyle(document.documentElement).getPropertyValue('--primary').trim() || '#22d3ee';
   640→   640→    const buf = new Uint8Array(512);
   641→   641→    let raf = 0;
   642→   642→    const draw = () => {
   643→   643→      raf = window.requestAnimationFrame(draw);
   644→   644→      lofiEngine.waveform(buf);
   645→   645→      c2d.clearRect(0, 0, w, h);
   646→   646→      c2d.beginPath();
   647→   647→      const step = w / (buf.length - 1);
   648→   648→      for (let i = 0; i < buf.length; i++) {
   649→   649→        const y = h / 2 + ((buf[i] - 128) / 128) * (h / 2) * 0.92;
   650→   650→        if (i === 0) c2d.moveTo(0, y);
   651→   651→        else c2d.lineTo(i * step, y);
   652→   652→      }
   653→   653→      c2d.strokeStyle = primary;
   654→   654→      c2d.lineWidth = 1.5;
   655→   655→      c2d.shadowColor = primary;
   656→   656→      c2d.shadowBlur = 8;
   657→   657→      c2d.stroke();
   658→   658→    };
   659→   659→    draw();
   660→   660→    return () => window.cancelAnimationFrame(raf);
   661→   661→  }, [playing]);
   662→   662→
   663→   663→  const togglePlay = () => {
   664→   664→    if (playing) lofiEngine.pause();
   665→   665→    else {
   666→   666→      lofiEngine.unlock(); // first-gesture AudioContext unlock
   667→   667→      lofiEngine.play();
   668→   668→    }
   669→   669→    onPlayingChange(lofiEngine.isPlaying());
   670→   670→  };
   671→   671→
   672→   672→  const track = lofiEngine.getTrack();
   673→   673→
   674→   674→  return (
   675→   675→    <div className="flex flex-col gap-3">
   676→   676→      <div className="flex items-center gap-3">
   677→   677→        <button
   678→   678→          type="button"
   679→   679→          onClick={togglePlay}
   680→   680→          aria-label={playing ? 'Pause music' : 'Play music'}
   681→   681→          className="flex h-11 w-11 shrink-0 items-center justify-center rounded-full border border-white/15 bg-white/10 text-white transition hover:bg-white/15"
   682→   682→          style={{ boxShadow: '0 0 22px -6px var(--primary)' }}
   683→   683→        >
   684→   684→          {playing ? <Pause className="h-4 w-4" /> : <Play className="ml-0.5 h-4 w-4" />}
   685→   685→        </button>
   686→   686→        <div className="min-w-0 flex-1">
   687→   687→          <div className="truncate text-sm font-medium">{track.name}</div>
   688→   688→          <div className="truncate text-[10px] text-white/50">
   689→   689→            {track.artist} · {track.bpm} BPM · {track.key}
   690→   690→          </div>
   691→   691→        </div>
   692→   692→        <span className="glass-chip shrink-0 rounded-full px-2 py-0.5 text-[8px] uppercase tracking-[0.14em] text-white/60">
   693→   693→          procedural lo-fi
   694→   694→        </span>
   695→   695→      </div>
   696→   696→      <canvas ref={canvasRef} className="h-12 w-full rounded-xl" aria-hidden="true" />
   697→   697→      <div className="flex items-center gap-2">
   698→   698→        <Button
   699→   699→          size="icon" variant="ghost" className="h-7 w-7 rounded-full text-white/70 hover:bg-white/10 hover:text-white"
   700→   700→          onClick={() => {
   701→   701→            lofiEngine.prev();
   702→   702→            onPlayingChange(lofiEngine.isPlaying());
   703→   703→          }}
   704→   704→          aria-label="Previous track"
   705→   705→        >
   706→   706→          <SkipBack className="h-3.5 w-3.5" />
   707→   707→        </Button>
   708→   708→        <Button
   709→   709→          size="icon" variant="ghost" className="h-7 w-7 rounded-full text-white/70 hover:bg-white/10 hover:text-white"
   710→   710→          onClick={() => {
   711→   711→            lofiEngine.next();
   712→   712→            onPlayingChange(lofiEngine.isPlaying());
   713→   713→          }}
   714→   714→          aria-label="Next track"
   715→   715→        >
   716→   716→          <SkipForward className="h-3.5 w-3.5" />
